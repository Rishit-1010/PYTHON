def largest(numbers):
    largest = numbers[0]
    for num in numbers:
        if num > largest:
            largest = num
    return largest
numbers = [10, 25, 7, 18, 30]
print("Largest number =", largest(numbers))

if speed <= 70:
print("OK")
else:
points = (speed - 70) // 5
print("Demerit Points =", points)

if points > 10:
print("License Suspended")
elif points > 5:
print("Warning")

speed = int(input("Enter speed: "))
check_speed(speed)

def calculate_bill(units):
if units <= 100:
bill = units * 4
elif units <= 200:
bill = 100 * 4 + (units - 100) * 6
else:
bill = 100 * 4 + 100 * 6 + (units - 200) * 8

bill = bill + 100

if bill > 2000:
bill = bill + bill * 5 / 100

print("Electricity Bill =", bill)

calculate_bill(units)
